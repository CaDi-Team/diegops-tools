pub enum DevtoolsCommand {
    /// Git identity setup
    Git {
        #[command(subcommand)]
        cmd: GitCommand,
    },
    /// GPG key management and git commit signing
    Gpg {
        #[command(subcommand)]
        cmd: GpgCommand,
    },
    /// SSH key management
    Ssh {
        #[command(subcommand)]
        cmd: SshCommand,
    },
}

/// Git sub-commands.
#[derive(clap::Subcommand)]
pub enum GitCommand {
    /// Set git global user.name and user.email
    Set {
        /// Git user name (overrides GitHub API lookup)
        #[arg(long)]
        name: Option<String>,
        /// Git user email (overrides GitHub API lookup)
        #[arg(long)]
        email: Option<String>,
    },
}

/// GPG sub-commands.
#[derive(clap::Subcommand)]
pub enum GpgCommand {
    /// Generate GPG identity config at ~/.diegops/gpg-config.yaml
    Init {
        /// Name for the GPG key (overrides git config / GitHub API)
        #[arg(long)]
        name: Option<String>,
        /// Email for the GPG key (overrides git config / GitHub API)
        #[arg(long)]
        email: Option<String>,
    },
    /// Generate GPG key, upload to GitHub, and configure git signing
    Set,
    /// Restart gpg-agent (fixes signing errors after system restart)
    Restart,
}

/// SSH sub-commands.
#[derive(clap::Subcommand)]
pub enum SshCommand {
    /// List local SSH keys and GitHub registered keys
    List,
    /// Print ~/.ssh/config contents
    Config,
    /// Create a new SSH key
    Create {
        /// Key name (default: id_<type>)
        #[arg(long)]
        name: Option<String>,
        /// Key type: ed25519 or rsa (default: ed25519)
        #[arg(long, default_value = "ed25519")]
        r#type: String,
        /// Email comment for the key
        #[arg(long)]
        email: Option<String>,
    },
}

/// Resolved identity (name + email).
pub struct Identity {
    pub name: String,
    pub email: String,
}

/// Resolves identity from: explicit flags > git config > gh API.
///
/// Returns an error with a user-friendly message if identity cannot be determined.
pub fn resolve_identity(
    name_flag: Option<&str>,
    email_flag: Option<&str>,
) -> Result<Identity, Box<dyn std::error::Error>> {
    let name = if let Some(n) = name_flag {
        n.to_owned()
    } else if let Some(n) = git_config_get("user.name")? {
        n
    } else {
        gh_api_login()?
    };

    let email = if let Some(e) = email_flag {
        e.to_owned()
    } else if let Some(e) = git_config_get("user.email")? {
        e
    } else {
        gh_api_primary_email()?
    };

    Ok(Identity { name, email })
}

/// Reads a git config value. Returns None if not set.
fn git_config_get(key: &str) -> Result<Option<String>, Box<dyn std::error::Error>> {
    let output = Command::new("git")
        .args(["config", "--global", key])
        .output();

    match output {
        Ok(o) if o.status.success() => {
            let val = String::from_utf8_lossy(&o.stdout).trim().to_owned();
            if val.is_empty() {
                Ok(None)
            } else {
                Ok(Some(val))
            }
        }
        _ => Ok(None),
    }
}

/// Fetches GitHub username via gh API.
fn gh_api_login() -> Result<String, Box<dyn std::error::Error>> {
    let output = Command::new("gh")
        .args(["api", "user", "--jq", ".login"])
        .output()
        .map_err(|_| "gh CLI not found — provide --name flag, or install gh from https://cli.github.com/")?;

    if !output.status.success() {
        return Err("gh is not authenticated. Run 'gh auth login' or provide --name flag".into());
    }

    let login = String::from_utf8_lossy(&output.stdout).trim().to_owned();
    if login.is_empty() {
        return Err("could not fetch username from GitHub API".into());
    }
    Ok(login)
}

/// Fetches primary email via gh API.
fn gh_api_primary_email() -> Result<String, Box<dyn std::error::Error>> {
    let output = Command::new("gh")
        .args(["api", "user/emails", "--jq", "[.[] | select(.primary==true)] | .[0].email"])
        .output()
        .map_err(|_| "gh CLI not found — provide --email flag, or install gh from https://cli.github.com/")?;

    if !output.status.success() {
        return Err("gh is not authenticated. Run 'gh auth login' or provide --email flag".into());
    }

    let email = String::from_utf8_lossy(&output.stdout).trim().to_owned();
    if email.is_empty() || email == "null" {
        return Err("could not fetch primary email from GitHub API. Provide --email flag".into());
    }
    Ok(email)
}

/// Checks that a command exists on PATH.
pub fn require_cmd(name: &str, install_hint: &str) -> Result<(), Box<dyn std::error::Error>> {
    match Command::new(name).arg("--version").output() {
        Ok(o) if o.status.success() => Ok(()),
        _ => Err(format!("{name} not found on PATH. {install_hint}").into()),
    }
}

//! Git identity setup.

use std::process::Command;

/// Sets git global user.name and user.email.
///
/// Resolution: --name/--email flags > gh API.
/// Idempotent: skips if already set to the same value.
pub fn set(name: Option<&str>, email: Option<&str>) -> Result<(), Box<dyn std::error::Error>> {
    super::devtools::require_cmd("git", "Install it from https://git-scm.com/")?;

    let identity = super::devtools::resolve_identity(name, email)?;

    let current_name = git_config_get("user.name");
    let current_email = git_config_get("user.email");

    if current_name.as_deref() == Some(&identity.name) {
        eprintln!("  user.name already set to '{}'", identity.name);
    } else {
        git_config_set("user.name", &identity.name)?;
        println!("  user.name = {}", identity.name);
    }

    if current_email.as_deref() == Some(&identity.email) {
        eprintln!("  user.email already set to '{}'", identity.email);
    } else {
        git_config_set("user.email", &identity.email)?;
        println!("  user.email = {}", identity.email);
    }

    println!("Done.");
    Ok(())
}

/// Reads a git global config value.
fn git_config_get(key: &str) -> Option<String> {
    Command::new("git")
        .args(["config", "--global", key])
        .output()
        .ok()
        .filter(|o| o.status.success())
        .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_owned())
        .filter(|s| !s.is_empty())
}

/// Sets a git global config value.
fn git_config_set(key: &str, value: &str) -> Result<(), Box<dyn std::error::Error>> {
    let output = Command::new("git")
        .args(["config", "--global", key, value])
        .output()
        .map_err(|e| format!("could not run git: {e}"))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("failed to set git config {key}: {stderr}").into());
    }
    Ok(())
}

//! GPG key management and git commit signing setup.

use serde::Deserialize;
use std::fs;
use std::path::Path;
use std::process::Command;

/// GPG config structure.
#[derive(Deserialize)]
pub struct GpgConfig {
    pub name: String,
    pub email: String,
}

/// Generates GPG identity config.
pub fn init(name: Option<&str>, email: Option<&str>) -> Result<(), Box<dyn std::error::Error>> {
    let dir = super::common::diegops_dir()?;
    let identity = super::devtools::resolve_identity(name, email)?;
    init_to(&dir, &identity.name, &identity.email)
}

/// Internal: writes config to a specific directory (for testability).
fn init_to(dir: &Path, name: &str, email: &str) -> Result<(), Box<dyn std::error::Error>> {
    let config_path = dir.join("gpg-config.yaml");

    if config_path.exists() {
        println!("Config already exists: {}", config_path.display());
        return Ok(());
    }

    fs::create_dir_all(dir)?;
    let content = format!(
        "# diegops GPG identity configuration\n\
         # Used by `diegops devtools gpg set` to generate and configure GPG keys.\n\
         \n\
         name: \"{name}\"\n\
         email: \"{email}\"\n"
    );
    fs::write(&config_path, content)?;

    println!("Created: {}", config_path.display());
    println!("  name:  {name}");
    println!("  email: {email}");
    println!("Edit the file if needed, then run `diegops devtools gpg set`.");
    Ok(())
}

/// Loads GPG config from `~/.diegops/gpg-config.yaml`.
fn load_gpg_config() -> Result<GpgConfig, Box<dyn std::error::Error>> {
    let path = super::common::diegops_dir()?.join("gpg-config.yaml");
    if !path.exists() {
        return Err("gpg config not found. Run 'diegops devtools gpg init' first".into());
    }
    let content = fs::read_to_string(&path)?;
    let config: GpgConfig = serde_yaml::from_str(&content)?;
    Ok(config)
}

/// Generates GPG key, uploads to GitHub, configures git signing.
pub fn set() -> Result<(), Box<dyn std::error::Error>> {
    super::devtools::require_cmd("gpg", "Install it: gpg4win (Windows), brew install gnupg (macOS), apt install gnupg (Linux)")?;
    super::devtools::require_cmd("git", "Install it from https://git-scm.com/")?;
    super::devtools::require_cmd("gh", "Install it from https://cli.github.com/")?;

    let config = load_gpg_config()?;
    eprintln!("Identity: {} <{}>", config.name, config.email);

    // Find or create GPG key
    eprintln!("\n==> Looking for existing GPG key for <{}>...", config.email);
    let key_id = match find_gpg_key(&config.email)? {
        Some(id) => {
            eprintln!("  Found existing key: {id}");
            id
        }
        None => {
            eprintln!("  No existing key — generating...");
            generate_gpg_key(&config.name, &config.email)?;
            find_gpg_key(&config.email)?
                .ok_or("GPG key generation succeeded but key was not found afterwards")?
        }
    };

    // Export and upload to GitHub
    eprintln!("\n==> Checking GitHub GPG keys...");
    if is_key_on_github(&key_id)? {
        eprintln!("  Key {key_id} already registered on GitHub — skipping upload.");
    } else {
        eprintln!("  Uploading key to GitHub...");
        upload_gpg_key_to_github(&key_id)?;
        eprintln!("  Uploaded successfully.");
    }

    // Configure git
    eprintln!("\n==> Configuring git signing...");
    let gpg_bin = which_gpg()?;
    for (key, value) in [
        ("user.signingkey", key_id.as_str()),
        ("commit.gpgsign", "true"),
        ("tag.gpgsign", "true"),
        ("gpg.program", &gpg_bin),
    ] {
        let output = Command::new("git")
            .args(["config", "--global", key, value])
            .output()
            .map_err(|e| format!("could not run git: {e}"))?;
        if !output.status.success() {
            return Err(format!("failed to set git config {key}").into());
        }
    }

    println!("\nDone.");
    println!("  GPG key ID         : {key_id}");
    println!("  user.signingkey    = {key_id}");
    println!("  commit.gpgsign     = true");
    println!("  tag.gpgsign        = true");
    println!("  gpg.program        = {gpg_bin}");
    println!("\nAll future commits and tags will be signed automatically.");
    Ok(())
}

/// Restarts gpg-agent.
pub fn restart() -> Result<(), Box<dyn std::error::Error>> {
    super::devtools::require_cmd("gpgconf", "It usually ships with gpg — reinstall your GPG package")?;

    eprintln!("Killing gpg-agent...");
    let _ = Command::new("gpgconf").args(["--kill", "gpg-agent"]).output();

    eprintln!("Launching gpg-agent...");
    let output = Command::new("gpgconf")
        .args(["--launch", "gpg-agent"])
        .output()
        .map_err(|e| format!("could not run gpgconf: {e}"))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("could not restart gpg-agent: {stderr}").into());
    }

    // Verify
    let verify = Command::new("gpg-connect-agent")
        .arg("/bye")
        .output();

    match verify {
        Ok(o) if o.status.success() => {
            println!("gpg-agent restarted successfully.");
        }
        _ => {
            println!("gpg-agent launched but verification failed. Try signing a test commit.");
        }
    }
    Ok(())
}

/// Finds an existing GPG secret key for the given email.
fn find_gpg_key(email: &str) -> Result<Option<String>, Box<dyn std::error::Error>> {
    let output = Command::new("gpg")
        .args(["--list-secret-keys", "--keyid-format=long", email])
        .output()
        .map_err(|e| format!("could not run gpg: {e}"))?;

    if !output.status.success() {
        return Ok(None);
    }

    let stdout = String::from_utf8_lossy(&output.stdout);
    for line in stdout.lines() {
        if line.starts_with("sec") {
            // Line format: sec   rsa4096/KEYID 2026-03-24 ...
            if let Some(key_part) = line.split_whitespace().nth(1) {
                if let Some(id) = key_part.split('/').nth(1) {
                    return Ok(Some(id.to_owned()));
                }
            }
        }
    }
    Ok(None)
}

/// Generates a new GPG key.
fn generate_gpg_key(name: &str, email: &str) -> Result<(), Box<dyn std::error::Error>> {
    let batch = format!(
        "%no-protection\n\
         Key-Type: RSA\n\
         Key-Length: 4096\n\
         Subkey-Type: RSA\n\
         Subkey-Length: 4096\n\
         Name-Real: {name}\n\
         Name-Email: {email}\n\
         Expire-Date: 0\n"
    );

    let output = Command::new("gpg")
        .args(["--batch", "--gen-key"])
        .stdin(std::process::Stdio::piped())
        .stdout(std::process::Stdio::piped())
        .stderr(std::process::Stdio::piped())
        .spawn()
        .and_then(|mut child| {
            use std::io::Write;
            if let Some(ref mut stdin) = child.stdin {
                stdin.write_all(batch.as_bytes())?;
            }
            child.wait_with_output()
        })
        .map_err(|e| format!("could not run gpg: {e}"))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("GPG key generation failed: {stderr}").into());
    }
    Ok(())
}

/// Checks if a GPG key is already registered on GitHub.
fn is_key_on_github(key_id: &str) -> Result<bool, Box<dyn std::error::Error>> {
    let output = Command::new("gh")
        .args(["api", "user/gpg_keys"])
        .output()
        .map_err(|e| format!("could not run gh: {e}"))?;

    if !output.status.success() {
        return Err("failed to query GitHub GPG keys. Check your gh auth scopes".into());
    }

    let json: serde_json::Value = serde_json::from_slice(&output.stdout)?;
    if let Some(keys) = json.as_array() {
        for key in keys {
            if key["key_id"].as_str() == Some(key_id) {
                return Ok(true);
            }
        }
    }
    Ok(false)
}

/// Uploads a GPG public key to GitHub.
fn upload_gpg_key_to_github(key_id: &str) -> Result<(), Box<dyn std::error::Error>> {
    let export = Command::new("gpg")
        .args(["--armor", "--export", key_id])
        .output()
        .map_err(|e| format!("could not export GPG key: {e}"))?;

    if !export.status.success() {
        return Err(format!("failed to export GPG public key for {key_id}").into());
    }

    let public_key = String::from_utf8_lossy(&export.stdout);
    let upload = Command::new("gh")
        .args([
            "api", "user/gpg_keys",
            "--method", "POST",

/// Returns the path to the gpg binary.

        Command::new("where").arg("gpg").output()
    } else {
        Command::new("which").arg("gpg").output()
    };

    let output = output.map_err(|e| format!("could not find gpg path: {e}"))?;
    let path = String::from_utf8_lossy(&output.stdout)
        .lines()
        .next()
        .unwrap_or("gpg")
        .trim()
        .to_owned();
    Ok(path)
}

/// Extracts key info using ssh-keygen -l.
fn ssh_key_info(pub_path: &std::path::Path) -> Option<SshKeyInfo> {
    let output = Command::new("ssh-keygen")
        .args(["-l", "-f"])
        .arg(pub_path)
        .output()
        .ok()?;

    if !output.status.success() {
        return None;
    }

    let line = String::from_utf8_lossy(&output.stdout);
    let parts: Vec<&str> = line.trim().split_whitespace().collect();
    // Format: 256 SHA256:abc123... user@host (ED25519)
    if parts.len() >= 4 {
        let fingerprint = parts[1].to_owned();
        let key_type = parts
            .last()?
            .trim_matches(|c| c == '(' || c == ')')
            .to_lowercase();
        Some(SshKeyInfo {
            key_type,
            fingerprint,
        })
    } else {
        None
    }
}

struct GhSshKey {
    title: String,
    key_type: String,
    fingerprint: String,
    added: String,
}

/// Fetches SSH keys from GitHub.
fn gh_ssh_keys() -> Result<Vec<GhSshKey>, Box<dyn std::error::Error>> {
    let output = Command::new("gh")
        .args(["ssh-key", "list"])
        .output()
        .map_err(|e| format!("could not run gh: {e}"))?;

    if !output.status.success() {
        return Err("gh ssh-key list failed".into());
    }

    let mut keys = Vec::new();
    for line in output.stdout.lines() {
        let line = line?;
        let parts: Vec<&str> = line.split('\t').collect();
        if parts.len() >= 4 {
            keys.push(GhSshKey {
                title: parts[0].to_owned(),
                key_type: parts[1].to_owned(),
                fingerprint: parts[2].to_owned(),
                added: parts[3].to_owned(),
            });
        }
    }
    Ok(keys)
}

/// Creates a new SSH key.
pub fn create(
    name: Option<&str>,
    key_type: &str,
    email: Option<&str>,
) -> Result<(), Box<dyn std::error::Error>> {
    super::devtools::require_cmd("ssh-keygen", "Install OpenSSH")?;

    // Resolve email for comment
    let comment = match email {
        Some(e) => e.to_owned(),
        None => super::devtools::resolve_identity(None, None)
            .map(|id| id.email)
            .unwrap_or_else(|_| "diegops".to_owned()),
    };

    // Resolve key path
    let ssh_dir = super::common::home_dir()?.join(".ssh");
    fs::create_dir_all(&ssh_dir)?;

    let base_name = name.unwrap_or(&format!("id_{key_type}")).to_owned();
    let mut key_path = ssh_dir.join(&base_name);

    if key_path.exists() {
        let postfix = if std::io::stdin().is_terminal() {
            eprintln!(
                "Key '{}' already exists. Enter a postfix (e.g., 'work', 'personal'):",
                key_path.display()
            );
            let mut input = String::new();
            std::io::stdin().read_line(&mut input)?;
            let input = input.trim().to_owned();
            if input.is_empty() {
                generate_random_postfix()
            } else {
                input
            }
        } else {
            generate_random_postfix()
        };
        key_path = ssh_dir.join(format!("{base_name}_{postfix}"));
    }

    eprintln!("Generating {key_type} key at {}...", key_path.display());

    let output = Command::new("ssh-keygen")
        .args([
            "-t", key_type,
            "-C", &comment,
            "-f",
        ])
        .arg(&key_path)
        .args(["-N", ""])
        .output()
        .map_err(|e| format!("could not run ssh-keygen: {e}"))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("ssh-keygen failed: {stderr}").into());
    }

    let pub_path = key_path.with_extension("pub");
    let public_key = fs::read_to_string(&pub_path)?;

    println!("Created: {}", key_path.display());
    println!();
    println!("Public key (copy this):");
    println!("{}", public_key.trim());
    Ok(())
}

/// Generates a random 6-character alphanumeric postfix using system time.
fn generate_random_postfix() -> String {
    use std::time::{SystemTime, UNIX_EPOCH};
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .subsec_nanos();
    format!("{:06x}", nanos & 0xFFFFFF)
}
